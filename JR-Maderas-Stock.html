async function registrarMov(origen){
var prodIdEl = origen==='alejandro' ? 'ale-prod' : 'mov-prod';
var unidEl = origen==='alejandro' ? 'ale-unid' : 'mov-unid';
var piesEl = origen==='alejandro' ? 'ale-pies' : 'mov-pies';
var notasEl = origen==='alejandro' ? 'ale-cliente' : 'mov-notas';
var tipoEl = origen==='alejandro' ? null : 'mov-tipo';
var prodId = parseInt(document.getElementById(prodIdEl).value);
var unid = parseInt(document.getElementById(unidEl).value)||0;
var pies = parseFloat(document.getElementById(piesEl).value)||0;
var notas = document.getElementById(notasEl).value.trim();
var tipoAuto = ROL==='alejandro'?'venta_ale':ROL==='romero'?'venta_rom':ROL==='local'?'venta_loc':'venta_ale';
var tipo = origen==='alejandro' ? tipoAuto : document.getElementById(tipoEl).value;
if(!prodId){alert('Seleccioná un producto.');return;}
if(!unid&&!pies){alert('Ingresá la cantidad.');return;}
var stock = await getStock();
var p = stock.find(x=>x.id===prodId);
if(!p){alert('Producto no encontrado.');return;}
var valor=0;
if(p.esM2){
var m2mov = pies||unid;
valor = m2mov*p.precioMay;
if(tipo!=='ingreso') p.m2 = Math.max(0,p.m2-m2mov);
else p.m2 += m2mov;
} else {
valor = unid*p.piesUnid*p.precioMay;
if(tipo!=='ingreso') p.unid = Math.max(0,p.unid-unid);
else p.unid += unid;
}
var movs = await getMov();
movs.push({
id:Date.now(), prod:p.nombre, prodId:prodId,
tipo:tipo, unid:unid, pies:pies||(unid*p.piesUnid),
valor:valor, notas:notas, origen:origen,
fecha:new Date().toISOString()
});
await saveStock(stock);
await saveMov(movs);
// Reset
document.getElementById(unidEl).value='';
document.getElementById(piesEl).value='';
document.getElementById(notasEl).value='';
await renderAll();
alert('✅ Registrado: '+fmt(valor));
}
